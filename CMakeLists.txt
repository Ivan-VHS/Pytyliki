cmake_minimum_required(VERSION 3.16)
project(PytylikiProcessorWorkstation VERSION 0.8.0 LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

option(PYTYLIKI_ENABLE_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer (GCC/Clang)" OFF)

if(MSVC)
  add_compile_options(/W4 /sdl /experimental:c11atomics)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()

if(PYTYLIKI_ENABLE_SANITIZERS)
  if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
  else()
    message(FATAL_ERROR "PYTYLIKI_ENABLE_SANITIZERS requires GCC or Clang")
  endif()
endif()

add_library(pytyliki_core src/concat.c src/core.c src/processor.c src/dsp.c src/improvement.c src/hw_rules.c src/pipeline.c src/policy.c src/processor_catalog.c src/evolution.c src/apsia.c src/syntax_pie.c src/berry_core.c src/mtte.c src/ironclad_systems.c src/dsp_engine.c src/forge66.c src/cpu_suite.c src/trimatrix_autoforge.c src/metric_geometry.c src/compute_ops.c src/memory.c src/metric_registry.c src/architecture_atlas.c src/self_product.c src/embedded.c src/systems_accel.c src/qcs.c src/trace.c src/three_medium.c src/division.c src/atomic_quantum.c src/budget.c .veelsevul/runtime/appguard.c)
target_include_directories(pytyliki_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_include_directories(pytyliki_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/.veelsevul/runtime)

# Stable C11 ABI for explicitly configured FFI consumers.
set_target_properties(pytyliki_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(pytyliki_ffi SHARED src/interop.c)
target_link_libraries(pytyliki_ffi PRIVATE pytyliki_core)
target_include_directories(pytyliki_ffi PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_compile_definitions(pytyliki_ffi PRIVATE PYTYLIKI_FFI_BUILD)
set_target_properties(pytyliki_ffi PROPERTIES VERSION 1.0.0 SOVERSION 1)
install(TARGETS pytyliki_ffi RUNTIME DESTINATION bin LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
install(FILES src/interop.h src/division.h src/qcs.h src/self_product.h src/trace.h src/three_medium.h src/atomic_quantum.h src/budget.h DESTINATION include/pytyliki)
if(UNIX)
  target_link_libraries(pytyliki_core PUBLIC m)
endif()
